#include <stdio.h>

int main()
{
    int arr[5] = {18, 3, 25, 9, 12};
    int i, j, temp;

    for (i = 0; i < 4; i++)
    {
        for (j = 0; j < 4 - i; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }

        printf("Pass %d: ", i + 1);

        for (j = 0; j < 5; j++)
        {
            printf("%d ", arr[j]);
        }

        printf("\n");
    }

    return 0;
}
