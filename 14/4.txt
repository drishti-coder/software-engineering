#include <stdio.h>

int main()
{
    int time[6] = {35, 20, 45, 15, 30, 25};
    int i, j;
    int small;
    int temp;

    for (i = 0; i < 5; i++)
    {
        small = i;

        for (j = i + 1; j < 6; j++)
        {
            if (time[j] < time[small])
            {
                small = j;
            }
        }

        temp = time[i];
        time[i] = time[small];
        time[small] = temp;

        printf("After selection %d: ", i + 1);

        for (j = 0; j < 6; j++)
        {
            printf("%d ", time[j]);
        }

        printf("\n");
    }

    return 0;
}
