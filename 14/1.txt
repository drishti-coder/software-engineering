#include <stdio.h>
#include <string.h>

int linearSearch(char usernames[][20], int size, char target[])
{
    int i;

    for (i = 0; i < size; i++)
    {
        if (strcmp(usernames[i], target) == 0)
        {
            return i;
        }
    }

    return -1;
}

int main()
{
    char usernames[7][20] = {
        "rahul123",
        "drishti01",
        "virat18",
        "anushka22",
        "rohit45",
        "neha10",
        "aman99"
    };

    char target[20];
    int result;

    printf("Enter username to search: ");
    scanf("%s", target);

    result = linearSearch(usernames, 7, target);

    if (result != -1)
    {
        printf("Username found at index %d", result);
    }
    else
    {
        printf("Username not found");
    }

    return 0;
}
