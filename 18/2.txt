

#include <stdio.h>

void swapvalues(int *a,int *b){
    
    int temp;
    
    temp=*a;
    *a=*b;
    *b=temp;
}

int main()
{
  int score1=100;
  int score2=200;
  
  printf("Before swapping: score1=%d, score2=%d\n",score1,score2);
  
  swapvalues(&score1,&score2);
  
  printf("After swapping: score1=%d, score2=%d",score1,score2);
    return 0;
}
