#include <stdio.h>

int main()
{
    float earnings;
    float tax;

    printf("Enter monthly earnings: ");
    scanf("%f", &earnings);

    if (earnings < 10000)
    {
        tax = 0;
    }
    else if (earnings <= 25000)
    {
        tax = earnings * 10 / 100;
    }
    else
    {
        tax = earnings * 20 / 100;
    }

    printf("Tax to be paid = Rs. %.2f", tax);

    return 0;
}
