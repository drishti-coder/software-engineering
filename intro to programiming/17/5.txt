#include <stdio.h>
#include <string.h>

int main()
{
    char message[100];
    int n;
    int i;

    printf("Enter message: ");
    fgets(message, 100, stdin);

    printf("Enter number: ");
    scanf("%d", &n);

    if (strlen(message) > n)
    {
        for (i = 0; i < n; i++)
        {
            printf("%c", message[i]);
        }

        printf("...");
    }
    else
    {
        printf("%s", message);
    }

    return 0;
}
