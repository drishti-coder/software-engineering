code:

#include <stdio.h>
int main(){
    int age;
    char confirm;

    printf("Enter your age: ");
    scanf("%d",&age);

    printf("first time user?(Y/N): ");
    scanf(" %c",&confirm);

    if(age<22){
        printf("10%% Discount");
    }else if(confirm=='Y'){
        printf("20%% discount");
    }else if(age>=22){
        printf("5%%  discount");
    }else{
        printf("No discount");
    }
    return 0;
}
