#include <stdio.h>

int main()
{
    char team[] = "CSK";
    int matches = 14;
    float runrate = 8.5;
    char code = 'C';

    printf("Team: %s\n", team);
    printf("Matches: %d\n", matches);
    printf("Run Rate: %.2f\n", runrate);
    printf("Team Code: %c\n", code);

    printf("Team takes %d bytes\n", sizeof(team));
    printf("Matches takes %d bytes\n", sizeof(matches));
    printf("Run Rate takes %d bytes\n", sizeof(runrate));
    printf("Team Code takes %d bytes\n", sizeof(code));

    return 0;
}
