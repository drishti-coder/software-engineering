

#include <stdio.h>
#include <stdbool.h>

struct fooditems{
    char name[50];
    float price;
    bool isveg;
};

int main()
{
   struct fooditems items[3]={
       {"Pizza",250.00,true},
       {"Burger",90.25,false},
       {"Fries",153.89,true}
   };
   
   for(int i=0; i<3; i++){
       if(items[i].isveg==true){
           printf("%s-%.2f\n",items[i].name,items[i].price);
       }
   }
   
  
 
 
    return 0;
}
