
#include <stdio.h>

struct Movie
{
    char name[50];
    char genre[50];
    float rating;
};

void displayMovie(struct Movie m)
{
    printf("\n----- Movie Details -----\n");
    printf("Movie Name: %s\n", m.name);
    printf("Genre: %s\n", m.genre);
    printf("Rating: %.1f/10\n", m.rating);
}

int main()
{
    struct Movie movie;

    printf("Enter movie name: ");
    scanf(" %[^\n]", movie.name);

    printf("Enter genre: ");
    scanf(" %[^\n]", movie.genre);

    printf("Enter rating: ");
    scanf("%f", &movie.rating);

    displayMovie(movie);

    return 0;
}
