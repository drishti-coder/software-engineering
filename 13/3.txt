#include <stdio.h>

int main()
{
    int orders[7];
    int i;
    int max;
    int day;

    printf("Enter orders for 7 days:\n");

    for (i = 0; i < 7; i++)
    {
        scanf("%d", &orders[i]);
    }

    max = orders[0];
    day = 0;

    for (i = 1; i < 7; i++)
    {
        if (orders[i] > max)
        {
            max = orders[i];
            day = i;
        }
    }

    printf("Highest orders = %d\n", max);
    printf("Day index = %d", day);

    return 0;
}
