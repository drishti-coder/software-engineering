#include <stdio.h>

int main()
{
    int orders[3][3] = {
        {20, 30, 25},
        {15, 25, 35},
        {40, 20, 30}
    };

    int transpose[3][3];

    int i, j;

    for (i = 0; i < 3; i++)
    {
        for (j = 0; j < 3; j++)
        {
            transpose[j][i] = orders[i][j];
        }
    }

    printf("Transpose matrix:\n");

    for (i = 0; i < 3; i++)
    {
        for (j = 0; j < 3; j++)
        {
            printf("%d ", transpose[i][j]);
        }

        printf("\n");
    }

    return 0;
}
