#include <stdio.h>

int main()
{
    int runs[3][4] = {
        {180, 200, 150, 170},
        {210, 160, 190, 200},
        {140, 175, 160, 185}
    };

    int i, j;
    int total;

    for (i = 0; i < 3; i++)
    {
        total = 0;

        for (j = 0; j < 4; j++)
        {
            total = total + runs[i][j];
        }

        printf("Team %d total runs = %d\n", i + 1, total);
    }

    return 0;
}
