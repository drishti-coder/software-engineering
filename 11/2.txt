#include <stdio.h>

int calculateCartTotal(int itemsArray[], int size)
{
    int total = 0;
    int i;

    for (i = 0; i < size; i++)
    {
        total = total + itemsArray[i];
    }

    return total;
}

int main()
{
    int itemsArray[3] = {299, 499, 199};
    int total;

    total = calculateCartTotal(itemsArray, 3);

    printf("Cart total = %d", total);

    return 0;
}
