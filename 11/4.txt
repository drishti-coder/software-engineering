#include <stdio.h>

void formatFollowerCount(int count)
{
    if (count < 1000)
    {
        printf("%d", count);
    }
    else if (count < 1000000)
    {
        printf("%.1fK", count / 1000.0);
    }
    else
    {
        printf("%.1fM", count / 1000000.0);
    }
}

int main()
{
    int count;

    printf("Enter follower count: ");
    scanf("%d", &count);

    formatFollowerCount(count);

    return 0;
}
