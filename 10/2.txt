#include <stdio.h>

int factorial(int no){
    
    int fact=1;
    int i;
    
    for(int i=1; i<=no; i++){
        fact=fact*i;
    }
     return fact;
}



int main()
{ int a;
  int ans;

    printf("Enter a no: ");
    scanf("%d",&a);
    
    ans=factorial(a);
    
    printf("factorial=%d",ans);
    
    
   
    return 0;
}
